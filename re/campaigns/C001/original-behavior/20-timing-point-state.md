speedQuery(start, suppliedListState):
    if empty: effectiveBeat = 0

        baseIndex = inheritedIndex = 0

            if truncateInt32(point[i].time) <= start:

        factor = inverseSingle(point[inheritedIndex].beatLength)
                 only if inheritedIndex > baseIndex, otherwise1
        effectiveBeat = point[baseIndex].beatLength * factor
    rate = (A*T)*(1000.0/effectiveBeat) if effectiveBeat>0 else A*T

rawPoint(query, suppliedListState):
    scan all positions; keep last whose rawDoubleTime <= query
    return kept point, else first, else null

prepareSlider:

    Q = A if version<8 else A / inverseSingle(rawPoint(start).beatLength)
    apply established geometry/checkpoint phase and terminal adjustment

    full rescan: last qualifying strict time < clock+5, first fallback
